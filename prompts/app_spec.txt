<project_specification>
  <project_name>CryptoTrader</project_name>

  <overview>
    A web-based AI-powered cryptocurrency trading platform that acts like a real human trader - analyzing markets in real-time, proposing and testing strategies via paper trading, and executing live trades with user approval. The system features a multi-agent architecture where specialized AI agents handle market analysis, strategy optimization, sentiment monitoring, risk management, and trade execution. Built for personal use with long-term continual AI integration.
  </overview>

      <state_management>React Context or Redux for complex state</state_management>
    </frontend>
    <backend>

  <architecture>
    <pattern>Multi-Agent Architecture (built from start, modular for evolution)</pattern>
    <agents>
      <agent name="Market Analyst Agent">
        <purpose>Monitors real-time price data, order books, indicators</purpose>
        <responsibilities>
          - Detect patterns, support/resistance levels
          - Feed insights to main trading AI
          - Continuous background monitoring
        </responsibilities>
      </agent>
      <agent name="Strategy Optimizer Agent">
        <purpose>Runs paper trade simulations and tunes parameters</purpose>
        <responsibilities>
          - Test strategies via paper trading
          - Tune entry/exit thresholds, position sizing, stop-loss/take-profit
          - Analyze strategy performance metrics
          - Propose strategy improvements
        </responsibilities>
      </agent>
      <agent name="Sentiment/News Agent">
        <purpose>Monitors external data sources for market sentiment</purpose>
        <responsibilities>
          - Monitor Twitter/X, Reddit, news feeds
          - Track on-chain data (whale movements, exchange flows)
          - Summarize relevant market sentiment
          - Flag unusual activity or breaking news
        </responsibilities>
      </agent>
      <agent name="Risk Monitor Agent">
        <purpose>Continuously calculates and monitors risk</purpose>
        <responsibilities>
          - Calculate overall risk score from multiple factors
          - Watch for threshold breaches
          - Trigger emergency actions (pause trading, alerts)
          - Safety-critical, runs independently
        </responsibilities>
      </agent>
      <agent name="Trade Executor Agent">
        <purpose>Handles actual order placement</purpose>
        <responsibilities>
          - Place orders with Kraken API
          - Manage order states and retries on failure
          - Log all actions (distinguish AI trades vs manual)
          - Reliable, isolated from other logic
        </responsibilities>
      </agent>
      <agent name="Orchestrator / Main AI">
        <purpose>Coordinates all agents and user interaction</purpose>
        <responsibilities>
          - Receive inputs from all specialized agents
          - Make high-level trading decisions
          - Handle user chat and collaboration
          - Manage strategy lifecycle (paper to live)
        </responsibilities>
      </agent>
    </agents>
    <communication>
      <method>Message queue (Redis) or database-based task queue</method>
      <conflict_resolution>Priority system with main orchestrator having final say</conflict_resolution>
    </communication>
  </architecture>
