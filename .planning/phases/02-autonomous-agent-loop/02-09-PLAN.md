export const agentsAPI = {

  dashboard: (pipelineLimit = 20) =>
    api.get('/api/agents/dashboard', { params: { pipeline_limit: pipelineLimit } }),

  /** Get status of all agents. */
  allStatus: () =>

  /** Get status of a single agent. */
  agentStatus: (agentName) =>

  /** Pause or resume an agent. action must be "pause" or "resume". */
  controlAgent: (agentName, action) =>

  /** Flush all messages from a queue channel. */
  flushQueue: (channel) =>
    api.post('/api/agents/queue/flush', { channel }),

  /** Retry a failed trade signal. */
  retrySignal: (signalId) =>
    api.post(`/api/agents/signals/${signalId}/retry`),
};

**1. `frontend/src/components/AgentStatusGrid.js`**

A grid of agent status cards with pause/resume toggle buttons.

Props: `{ agents, onControl, controlLoading }`
- `agents`: Array of `{ name, description, running, paused, started_at, queue_size, last_heartbeat, heartbeat_age_seconds }`
- `onControl(agentName, action)`: callback for pause/resume
- `controlLoading`: string of agent name currently being controlled, or null

Each agent card displays:
- Agent name (formatted: replace underscores with spaces, title case) and description
- Status indicator: green dot for running, yellow dot for paused, red dot for stopped
- Heartbeat: show `heartbeat_age_seconds` rounded to 1 decimal as "X.Xs ago", color-coded (green <10s, yellow 10-30s, red >30s)
- Queue size badge
- Pause/Resume toggle button (uses `onControl`). Button shows spinner/disabled when `controlLoading === agent.name`. Pause button is amber, Resume button is emerald.

Layout: CSS grid, `grid-cols-1 md:grid-cols-2 lg:grid-cols-3` to fit 3 agents across on large screens.

**2. `frontend/src/components/QueueMetrics.js`**

Displays queue depth, throughput, and channel breakdown.

Props: `{ metrics }`
- `metrics`: `{ channels, total_depth, throughput_per_minute }`

Display:
- Total queue depth as a large number with color coding (green <20, yellow 20-80, red >80)
- Per-channel breakdown in a compact table or list (channel name, depth per priority, total)

**3. `frontend/src/components/PipelineTimeline.js`**

Shows recent pipeline events as a vertical timeline.

Props: `{ events }`
- `events`: Array of `{ timestamp, source_agent, target_agent, event_type, channel, priority, summary, metadata }`

- Timestamp (relative, e.g., "2s ago", "1m ago") using simple `Date.now() - Date.parse(timestamp)` math

- Event type badge (color-coded: insight=blue, signal=amber, order=emerald)

Timeline uses a left-border line pattern: `border-l-2 border-gray-700 pl-4 ml-2` per entry with a small dot marker.

3. **Add data fetching with polling (5-second interval per research recommendation):**
   ```javascript
   useEffect(() => {
     const controller = new AbortController();
     let intervalId;

     const fetchAgentDashboard = async () => {
       try {
         const response = await agentsAPI.dashboard(20);
         if (!controller.signal.aborted) {
           setAgentData(response.data);
           setAgentError('');
         }
       } catch (err) {
         if (!controller.signal.aborted) {
           console.error('Failed to fetch agent dashboard:', err);
           setAgentError(err?.message || 'Unable to load agent status.');
         }
       } finally {
         if (!controller.signal.aborted) {
           setAgentLoading(false);
         }
       }
     };

     fetchAgentDashboard();
     intervalId = setInterval(fetchAgentDashboard, 5000);

     return () => {
       controller.abort();
       clearInterval(intervalId);
     };
   }, []);
   ```

4. **Add handler functions:**
   ```javascript
   const handleAgentControl = async (agentName, action) => {
     setControlLoading(agentName);
     try {
       await agentsAPI.controlAgent(agentName, action);
       // Refresh dashboard data immediately after control action
       const response = await agentsAPI.dashboard(20);
       setAgentData(response.data);
     } catch (err) {
       console.error(`Failed to ${action} agent ${agentName}:`, err);
     } finally {
       setControlLoading(null);
     }
   };

   const handleFlushQueue = async (channel) => {
     if (!window.confirm(`Flush all messages from "${channel}"? This cannot be undone.`)) return;
     setFlushLoading(true);
     try {
       await agentsAPI.flushQueue(channel);
       const response = await agentsAPI.dashboard(20);
       setAgentData(response.data);
     } catch (err) {
       console.error('Failed to flush queue:', err);
     } finally {
       setFlushLoading(false);
     }
   };

   const handleRetrySignal = async (signalId) => {
     try {
       await agentsAPI.retrySignal(signalId);
       const response = await agentsAPI.dashboard(20);
       setAgentData(response.data);
     } catch (err) {
       console.error('Failed to retry signal:', err);
     }
   };
   ```
