3. **Add _apply_fallback_strategy() method:**
   ```python
   async def _apply_fallback_strategy(
       self, signal: TradeSignal, pending: PendingOrder
   ) -> Optional[List[str]]:
       """
       Apply fallback strategy when primary order fails.

       Per user decision: "apply a fallback strategy automatically
       (e.g., reduced position size or alternate venue) before marking
       the signal failed."

       Returns order_ids if fallback succeeds, None otherwise.
       """
       # Save original volume on first fallback
       if pending.original_volume is None:
           pending.original_volume = signal.volume

       # Check if we've exhausted fallback attempts
       if pending.fallback_attempts >= self.MAX_FALLBACK_ATTEMPTS:
           self._log_system_event(
               "error",
               "Fallback strategy exhausted",
               {
                   "signal_id": signal.signal_id,
                   "original_volume": str(pending.original_volume),
                   "final_volume": str(signal.volume),
                   "fallback_attempts": pending.fallback_attempts,
                   "last_error": pending.last_error,
               },
           )
           return None

       # Calculate reduced volume
       reduced_volume = signal.volume * self.FALLBACK_VOLUME_REDUCTION

       # Check minimum volume threshold
       if reduced_volume < self.MIN_FALLBACK_VOLUME:
           self._log_system_event(
               "warning",
               "Fallback volume below minimum threshold",
               {
                   "signal_id": signal.signal_id,
                   "reduced_volume": str(reduced_volume),
                   "min_volume": str(self.MIN_FALLBACK_VOLUME),
               },
           )
           return None

       # Update signal with reduced volume (create new signal to preserve immutability)
       # Actually, modify the existing signal for the retry
       original_volume = signal.volume
       signal.volume = reduced_volume
       pending.fallback_attempts += 1

       self._log_system_event(
           "info",
           "Applying fallback: reduced position size",
           {
               "signal_id": signal.signal_id,
               "original_volume": str(original_volume),
               "reduced_volume": str(reduced_volume),
               "fallback_attempt": pending.fallback_attempts,
           },
       )

       # Retry with reduced volume
       return await self._place_order_with_retries(signal, pending)
   ```
